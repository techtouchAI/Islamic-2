    43	class IslamicPatternPainter extends CustomPainter {
    44	  final Color color;
    45	  IslamicPatternPainter({required this.color});
    46
    47	  @override
    48	  void paint(Canvas canvas, Size size) {
    49	    final paint = Paint()
    50	      ..color = color.withValues(alpha: 0.1)
    51	      ..strokeWidth = 0.8
    52	      ..style = PaintingStyle.stroke;
    53
    54	    const double step = 60;
    55	    for (double x = 0; x < size.width + step; x += step) {
    56	      for (double y = 0; y < size.height + step; y += step) {
    57	        final center = Offset(x, y);
    58	        _drawStar(canvas, center, step * 0.4, paint);
    59	        canvas.drawCircle(center, step * 0.1, paint);
    60	      }
    61	    }
    62	  }
    63
    64	  void _drawStar(Canvas canvas, Offset center, double radius, Paint paint) {
    65	    final path = Path();
    66	    for (int i = 0; i < 8; i++) {
    67	      double angle = (i * 45) * pi / 180;
    68	      double x = center.dx + radius * cos(angle);
    69	      double y = center.dy + radius * sin(angle);
    70	      if (i == 0) {
    71	        path.moveTo(x, y);
    72	      } else {
    73	        path.lineTo(x, y);
    74	      }
    75	      double nextAngle = (i * 45 + 22.5) * pi / 180;
    76	      double nextX = center.dx + (radius * 0.7) * cos(nextAngle);
    77	      double nextY = center.dy + (radius * 0.7) * sin(nextAngle);
    78	      path.lineTo(nextX, nextY);
    79	    }
    80	    path.close();
    81	    canvas.drawPath(path, paint);
    82	  }
    83
    84	  @override
    85	  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
    86	}

   115	Widget buildImage(String? path, {double? height, BoxFit fit = BoxFit.contain}) {
   116	  if (path == null || path.isEmpty) {
   117	    return const SizedBox();
   118	  }
   119	  if (path.startsWith('data:image')) {
   120	    try {
   121	      final bytes = Uri.parse(path).data!.contentAsBytes();
   122	      return Image.memory(
   123	        Uint8List.fromList(bytes),
   124	        height: height,
   125	        fit: fit,
   126	        errorBuilder: (c, e, s) => const Icon(Icons.broken_image),
   127	      );
   128	    } catch (e) {
   129	      return const Icon(Icons.broken_image);
   130	    }
   131	  }
   132	  if (path.startsWith('https://')) {
   133	    return Image.network(
   134	      path,
   135	      height: height,
   136	      fit: fit,
   137	      errorBuilder: (c, e, s) => const Icon(Icons.error),
   138	    );
   139	  }
   140	  return Image.asset(
   141	    path,
   142	    height: height,
   143	    fit: fit,
   144	    errorBuilder: (c, e, s) => const Icon(Icons.image_not_supported),
   145	  );
   146	}

   217	  Future<void> _loadSettings() async {
   218	    final prefs = await SharedPreferences.getInstance();
   219	    final dbSettings = DataManager.getSettings();
   220	    int defaultPrimary =
   221	        int.tryParse(dbSettings['primary_color'] ?? '0xFF2196F3') ?? 0xFF2196F3;
   222	    int defaultCard =
   223	        int.tryParse(dbSettings['card_color'] ?? '0xFFFFFFFF') ?? 0xFFFFFFFF;
   224	    setState(() {
   225	      _themeMode = (prefs.getString('theme') ?? 'light') == 'light'
   226	          ? ThemeMode.light
   227	          : ThemeMode.dark;
   228	      _fontSizeFactor = prefs.getDouble('fontSize') ?? 1.0;
   229	      _primaryColor = Color(prefs.getInt('primaryColor') ?? defaultPrimary);
   230	      _uiOpacity = prefs.getDouble('uiOpacity') ??
   231	          (dbSettings['ui_opacity']?.toDouble() ?? 1.0);
   232	      _backgroundImagePath = prefs.getString('backgroundImage');
   233	      _selectedBase64Bg = prefs.getString('custom_bg_base64_selected');
   234	      _cardColor = Color(prefs.getInt('cardColor') ?? defaultCard);
   235	      _hijriAdjustment = prefs.getInt('hijri.date.correction.value') ?? 0;
   236	      final sections = DataManager.getSections();
   237	      final allSections = {
   238	        ...sections,
   239	        'hadith': {},
   240	        'names_allah': {},
   241	        'adhkar': {},
   242	      };
   243	      _homeVisibility = {};
   244	      allSections.forEach((key, value) {
   245	        _homeVisibility[key] =
   246	            prefs.getBool('vis_$key') ?? (value['visible_home'] ?? true);
   247	      });
   248	      _homeVisibility['inspiration'] = prefs.getBool('vis_inspiration') ??
   249	          (dbSettings['show_inspiration'] ?? true);
   250	      _homeVisibility['day_dua'] = prefs.getBool('vis_day_dua') ?? true;
   251	    });
   252	  }

   254	  Future<void> _saveSetting(String key, dynamic value) async {
   255	    final prefs = await SharedPreferences.getInstance();
   256	    if (value is String) prefs.setString(key, value);
   257	    if (value is double) prefs.setDouble(key, value);
   258	    if (value is int) prefs.setInt(key, value);
   259	  }

   950	  void _updateCurrentPrayer() {

   955	      ..sort((a, b) => a.value.compareTo(b.value));

  1021	  void _loadDailyDua() {
  1022	    final now = DateTime.now();
  1023	    final dateKey = intl.DateFormat('yyyy-MM-dd').format(now);
  1024	    final cacheKey = "${dateKey}_${DataManager.dbNotifier.value}";
  1025
  1026	    if (_cachedDuaKey == cacheKey) {
  1027	      _inspirationDua = _cachedInspirationDua;
  1028	      _dayDua = _cachedDayDua;
  1029	      return;
  1030	    }
  1031
  1032	    final dayOfYear = int.parse(intl.DateFormat('D').format(now));
  1033	    _inspirationDua =
  1034	        DailyDuas.shortDuas[dayOfYear % DailyDuas.shortDuas.length];
  1035
  1036	    final dayNameAr = intl.DateFormat('EEEE', 'ar_SA').format(now);
  1037	    final allDaysDuas = DataManager.getItems('duas_days');
  1038
  1039	    final normalizedDay = dayNameAr.normalizeArabic();
  1040
  1041	    final itemsForToday = allDaysDuas.where((it) {
  1042	      if (it['_normalized_title'] == null) {
  1043	        it['_normalized_title'] = it['title'].toString().normalizeArabic();
  1044	      }
  1045	      final title = it['_normalized_title'] as String;
  1046	      return title.contains(normalizedDay);
  1047	    }).toList();
  1048
  1049	    if (itemsForToday.isNotEmpty) {
  1050	      String combinedTitle = "أعمال يوم $dayNameAr";
  1051	      StringBuffer combinedContent = StringBuffer();
  1052	      for (var it in itemsForToday) {
  1053	        combinedContent.writeln("✨ ${it['title']} ✨");
  1054	        combinedContent.writeln("${it['content']}");
  1055	        combinedContent.writeln("");
  1056	      }
  1057	      _dayDua = {
  1058	        "title": combinedTitle,
  1059	        "content": combinedContent.toString().trim(),
  1060	      };
  1061	    } else {
  1062	      _dayDua = null;
  1063	    }
  1064
  1065	    _cachedDuaKey = cacheKey;
  1066	    _cachedInspirationDua = _inspirationDua;
  1067	    _cachedDayDua = _dayDua;
  1068	  }
